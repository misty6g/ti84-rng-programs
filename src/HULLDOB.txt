ClrHome
Disp "HULL-DOBELL"
Input "M=",M
Input "A=",A
Input "C=",C
If M<2 or fPart(M) or fPart(A) or fPart(C)
Then
Disp "NEED INTEGERS","AND M≥2"
Stop
End
A-1→B
If C=0
Then
M→G
Else
gcd(abs(C),M)→G
End
(G=1)→P
{0}→ʟPF
0→J
M→N
1→Q
2→D
While D²≤N
If fPart(N/D)=0
Then
J+1→J
D→ʟPF(J)
If fPart(B/D)
0→Q
While fPart(N/D)=0
N/D→N
End
End
D+1+(D>2)→D
End
If N>1
Then
J+1→J
N→ʟPF(J)
If fPart(B/N)
0→Q
End
fPart(M/4) or not(fPart(B/4))→R
ClrHome
Disp "PRIMES OF M:",ʟPF
Disp "A-1=",B
Pause
ClrHome
Disp "1)GCD(C,M)=1","  GCD=",G,"  "+sub("FALSETRUE ",1+5P,5)
Disp "2)EVERY PRIME OF","  M DIVIDES A-1","  "+sub("FALSETRUE ",1+5Q,5)
Pause
ClrHome
Disp "3)IF 4 DIVIDES M","  THEN 4 DIV A-1"
If fPart(M/4)
Then
Disp "  (4 NOT DIV M)"
Else
Disp "  (4 DIVIDES M)"
End
Disp "  "+sub("FALSETRUE ",1+5R,5)
If P and Q and R
Then
Disp "FULL PERIOD","PERIOD = M =",M
Else
Disp "NOT FULL PERIOD","(PERIOD < M)"
End
