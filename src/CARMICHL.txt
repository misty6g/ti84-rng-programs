ClrHome
Menu("CARMICHAEL","ENTER M",1,"ENTER FACTORS",2)
Lbl 1
Input "M=",M
If M<1 or fPart(M)
Then
Disp "NEED INTEGER M≥1"
Stop
End
{0}→ʟP
{0}→ʟE
0→J
M→N
2→D
While D²≤N
If fPart(N/D)=0
Then
J+1→J
D→ʟP(J)
0→K
While fPart(N/D)=0
N/D→N
K+1→K
End
K→ʟE(J)
End
D+1+(D>2)→D
End
If N>1
Then
J+1→J
N→ʟP(J)
1→ʟE(J)
End
Goto 3
Lbl 2
ClrHome
Disp "EG M=2^3*5","PRIMES {2,5}","EXPS {3,1}"
Input "PRIMES:",ʟP
Input "EXPONENTS:",ʟE
dim(ʟP)→J
If J≠dim(ʟE)
Then
Disp "SIZE MISMATCH"
Stop
End
prod(ʟP^ʟE)→M
Lbl 3
ClrHome
If J=0
Then
Disp "M=1","LAMBDA(1)=1"
Stop
End
J→dim(ʟP)
J→dim(ʟE)
seq(0,I,1,J)→ʟLAM
1→L
Disp "M=",M,"{P,K,LAMBDA}"
For(I,1,J)
ʟP(I)→P
ʟE(I)→K
If P=2
Then
If K≤2
Then
K→Y
Else
2^(K-2)→Y
End
Else
P^(K-1)*(P-1)→Y
End
Y→ʟLAM(I)
lcm(L,Y)→L
Disp {P,K,Y}
If fPart(I/5)=0 and I<J
Pause
End
Pause
Disp "LAMBDA(M)=LCM=",L
